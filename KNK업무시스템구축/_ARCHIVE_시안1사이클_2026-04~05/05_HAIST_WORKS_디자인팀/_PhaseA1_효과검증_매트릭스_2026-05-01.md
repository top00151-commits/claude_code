```bash
# 시간 (수동 — 대표 셀프 보고)

# 라운드 (수동 — 캔버스 채팅 수)

# 5종 grep PASS율 (자동)
for c in 시각 토큰 구조 외부자산 접근성; do
  # 카테고리별 grep 5종 실행
  echo "C-$c: __/PASS|FAIL"
done

# 토큰 정합도 (자동)
total_var=$(grep -cE 'var\(--' [산출물])
total_hex=$(grep -cE '#[0-9a-fA-F]{3,6}' [산출물])
echo "정합도: $((total_var * 100 / (total_var + total_hex)))%"
```
